```powershell
python -m venv .venv
.\.venv\Scripts\python.exe -m pip install -r requirements.lock -r requirements-knowledge.lock
.\.venv\Scripts\python.exe scripts/verify_matter_update_inputs.py
.\.venv\Scripts\python.exe scripts/reproduce_matter_project.py --runtime .runtime/clean-reproduction
```
